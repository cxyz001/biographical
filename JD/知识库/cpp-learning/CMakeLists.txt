cmake_minimum_required(VERSION 3.16)
project(cpp_learning LANGUAGES CXX)
find_package(Threads REQUIRED)
add_library(learning_core INTERFACE)
target_include_directories(learning_core INTERFACE ${CMAKE_CURRENT_SOURCE_DIR}/include)
target_compile_features(learning_core INTERFACE cxx_std_17)
target_link_libraries(learning_core INTERFACE Threads::Threads)
foreach(name IN ITEMS telemetry_demo core_test)
    if(name STREQUAL "telemetry_demo")
        add_executable(${name} src/main.cpp)
    else()
        add_executable(${name} tests/core_test.cpp)
    endif()
    target_link_libraries(${name} PRIVATE learning_core)
    if(MSVC)
        target_compile_options(${name} PRIVATE /W4 /WX /permissive- /utf-8)
    else()
        target_compile_options(${name} PRIVATE -Wall -Wextra -Wpedantic -Werror)
    endif()
endforeach()
enable_testing()
add_test(NAME core_contracts COMMAND core_test)
set_tests_properties(core_contracts PROPERTIES TIMEOUT 20)
